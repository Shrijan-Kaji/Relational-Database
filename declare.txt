DECLARE
vn_counter NUMBER(3):=0;
BEGIN
LOOP
IF vn_counter>3 THEN
EXIT;
DBMS_OUTPUT_LINE('counter has not been conceded 3');
vn_counter:=vn_counter+1;
END LOOP;
END;




DECLARE
vn_counter NUMBER(3):=0;
BEGIN
LOOP
EXIT WHEN vn_counter>3;
DBMS_OUTPUT.PUT_LINE('counter hasn't exceeded 3');
vn_counter:=vn_counter+1;
END LOOP;
END;
